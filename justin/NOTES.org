* Building the SW env
#+begin_src  bash
pixi global install rattler-build rattler-index
mkdir -p ../conda-channel/noarch
rattler-index fs ../conda-channel
mkdir ../conda-recipes
cd ../conda-recipes/
pixi exec grayskull pypi https://github.com/larpix/adc64format
pixi exec grayskull pypi https://github.com/DUNE/h5flow
pixi exec grayskull pypi https://github.com/DUNE/ndlar_flow
rattler-build build --recipe adc64format/recipe.yaml --test skip --recipe adc64format/recipe.yaml --output-dir ../conda-channel -c conda-forge
rattler-build build --recipe h5flow/recipe.yaml --test skip --recipe adc64format/recipe.yaml --output-dir ../conda-channel -c conda-forge
rattler-build build --recipe ndlar_flow/recipe.yaml --test skip --recipe adc64format/recipe.yaml --output-dir ../conda-channel -c conda-forge
pixi init nearline_2x2
cd nearline_2x2
pixi add 'python=3.14.*'
realpath ../../conda-channel/
pixi add adc64format
pixi add h5flow
pixi add ndlar_flow
pixi pack
mv environment.tar nearline_2x2_v0.0.0.tar 
justin-cvmfs-upload nearline_2x2_v0.0.0.tar 
#+end_src
