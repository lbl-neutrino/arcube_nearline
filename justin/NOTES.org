#+begin_src bash
#!/bin/bash

did_pfn_rse=$("$JUSTIN_PATH/justin-get-file")

if [[ $? != 0 ]]; then
    echo "Got nothing from justin-get-file; bailing"
    exit
fi

echo "$did_pfn_rse"
did=$(echo $did_pfn_rse | cut -f1 -d' ')

hello=hello_grid_run3.text
echo "Hello from $(hostname)!" > $hello

# temporary till the lab signs a new deal with sectigo
cacert=/cvmfs/fifeuser1.opensciencegrid.org/sw/dune/e95c8a256a0133d467c5257ad7555de9cd9937a6/labpix.crt
auth=(--cacert "$cacert" --cert "$X509_USER_PROXY" --key "$X509_USER_PROXY")

curl "${auth[@]}" --upload-file $hello https://labpix.lbl.gov:55556/$hello

echo "$did" > justin-processed-dids.txt
#+end_src
